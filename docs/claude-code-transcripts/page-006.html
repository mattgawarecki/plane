Real gap — good catch. Right now Stop = hard cancel: terminal, drops in-flight work, no recovery. That fails your "recover from a mistake, non-disruptive" bar. The fix is two moves:

1. Make the one-click default non-destructive: Pause, not Stop. The contract already has pause/resume. So the common accidental click just suspends the run — the agent keeps its context/checkpoint (cheap to resume, non-disruptive to the agent) and recovery is a one-tap Resume. Destructive hard-Cancel becomes a deliberate secondary action, not the primary button.

2. Even hard-Cancel gets a brief Undo window (no modal — modals are the disruptive thing). The clean way: the client defers dispatching the cancel for ~5s and shows "Stopped · Undo"; if you undo, the command is never sent — the agent kept running the whole time, none the wiser. If the window lapses, it dispatches. That's maximally non-disruptive to both sides. (A backend soft-cancelling state is the more refresh-robust variant; note it for later.)

```tsx
import { useCallback, useEffect, useRef, useState } from "react";
import { useAgentStore } from "@/plane-web/hooks/agents/use-agent-store";

export const CANCEL_UNDO_MS = 5000;

export const useDeferredCancel = (workspaceId: string) => {
  const store = useAgentStore();
  const [pending, setPending] = useState<Set<string>>(new Set());
  const timers = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());

  const clear = useCallback((runId: string) => {
    const tm = timers.current.get(runId);
    if (tm) clearTimeout(tm);
    timers.current.delete(runId);
    setPending((prev) => {
      const next = new Set(prev);
      next.delete(runId);
      return next;
    });
  }, []);

  const requestCancel = useCallback(
    (runId: string) => {
      setPending((prev) => new Set(prev).add(runId));
      const tm = setTimeout(() => {
        clear(runId);
        void store.runCommand({ workspaceId, command: { runId, type: "cancel", payload: {} } });
      }, CANCEL_UNDO_MS);
      timers.current.set(runId, tm);
    },
    [clear, store, workspaceId]
  );

  const undoCancel = useCallback((runId: string) => clear(runId), [clear]); // command never sent

  useEffect(() => () => timers.current.forEach((tm) => clearTimeout(tm)), []);

  return { pending, requestCancel, undoCancel };
};
```
